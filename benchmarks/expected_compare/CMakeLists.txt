cmake_minimum_required(VERSION 3.23)
project(expected_comparison LANGUAGES CXX)

# Standalone comparison: dependency management stays outside the source tree.
set(CBOR_TAGS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CBOR_TAGS_DEBUG 0)
set(CBOR_TAGS_STL_ONLY 0)
set(CBOR_TAGS_USE_STD_EXPECTED 0)
set(CBOR_TAGS_USE_STD_REFLECTION 0)
set(CBOR_TAGS_USE_BOOST_PFR_NAMES 0)
set(CBOR_TAGS_USE_MAGIC_ENUM_NAMES 0)
configure_file("${CBOR_TAGS_ROOT}/cbor_tags_config.h.in" "${CMAKE_BINARY_DIR}/generated/cbor_tags/cbor_tags_config.h")

find_path(TL_EXPECTED_INCLUDE_DIR NAMES tl/expected.hpp DOC "Installed or externally cached tl::expected headers")
if(NOT TL_EXPECTED_INCLUDE_DIR)
  message(FATAL_ERROR "Set TL_EXPECTED_INCLUDE_DIR to an installed or externally cached tl::expected include directory.")
endif()
configure_file(tl_expected_backend.h "${CMAKE_BINARY_DIR}/tl/include/cbor_tags/expected.h" COPYONLY)

if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The relocation probes currently require GCC/Clang on Linux.")
endif()
set(CMP_OPTIMIZATION "-O3" CACHE STRING "Optimization flag shared by both backends")

foreach(backend IN ITEMS ours tl)
  foreach(kind IN ITEMS kernels expected_only)
    add_library(${kind}_${backend} OBJECT kernels.cpp)
    target_compile_options(${kind}_${backend} PRIVATE ${CMP_OPTIMIZATION} -DNDEBUG -fno-pie -ffunction-sections -fdata-sections)
    target_include_directories(${kind}_${backend} PUBLIC "${CMAKE_BINARY_DIR}/generated" "${CBOR_TAGS_ROOT}/include")
    if(backend STREQUAL "tl")
      target_include_directories(${kind}_${backend} BEFORE PUBLIC "${CMAKE_BINARY_DIR}/tl/include" "${TL_EXPECTED_INCLUDE_DIR}")
    endif()
  endforeach()
  target_compile_definitions(expected_only_${backend} PUBLIC CMP_EXPECTED_ONLY)

  add_executable(size_expected_${backend} size_driver.cpp)
  target_link_libraries(size_expected_${backend} PRIVATE expected_only_${backend})
  target_compile_options(size_expected_${backend} PRIVATE ${CMP_OPTIMIZATION} -DNDEBUG -fno-pie)
  target_link_options(size_expected_${backend} PRIVATE -no-pie -Wl,--gc-sections -Wl,--build-id=none)

  add_executable(size_${backend} size_driver.cpp)
  target_link_libraries(size_${backend} PRIVATE kernels_${backend})
  target_compile_options(size_${backend} PRIVATE ${CMP_OPTIMIZATION} -DNDEBUG -fno-pie)
  target_link_options(size_${backend} PRIVATE -no-pie -Wl,--gc-sections -Wl,--build-id=none)

  add_executable(layout_${backend} layout.cpp)
  target_link_libraries(layout_${backend} PRIVATE kernels_${backend})
  target_compile_options(layout_${backend} PRIVATE ${CMP_OPTIMIZATION} -DNDEBUG -fno-pie)
  target_link_options(layout_${backend} PRIVATE -no-pie -Wl,--gc-sections -Wl,--build-id=none)

  foreach(offset IN ITEMS 0 16 32 48)
    math(EXPR address "6291456 + ${offset}" OUTPUT_FORMAT HEXADECIMAL)
    add_executable(runtime_${backend}_${offset} EXCLUDE_FROM_ALL runtime.cpp)
    target_link_libraries(runtime_${backend}_${offset} PRIVATE kernels_${backend})
    target_compile_options(runtime_${backend}_${offset} PRIVATE ${CMP_OPTIMIZATION} -DNDEBUG -fno-pie)
    target_link_options(runtime_${backend}_${offset} PRIVATE -no-pie -Wl,--gc-sections -Wl,--build-id=none
                                           "-Wl,--section-start=.expected_cmp=${address}")
  endforeach()
endforeach()

enable_testing()
add_test(NAME expected_comparison_ours COMMAND size_ours)
add_test(NAME expected_comparison_tl COMMAND size_tl)
add_test(NAME expected_only_ours COMMAND size_expected_ours)
add_test(NAME expected_only_tl COMMAND size_expected_tl)
